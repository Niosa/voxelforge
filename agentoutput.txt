### `src/geo/validate.ts`

```ts
import type { TerraGeometry } from '@/entities/types';

export function isClosedRing(ring: number[][]): boolean {
  if (ring.length < 4) return false;
  const first = ring[^0];
  const last = ring[ring.length - 1];
  return first[^0] === last[^0] && first[^1] === last[^1];
}

export function validateGeometry(geometry: TerraGeometry): string[] {
  const errors: string[] = [];

  if (geometry.type === 'Point') {
    const [lon, lat] = geometry.coordinates;
    if (!Number.isFinite(lon) || !Number.isFinite(lat)) {
      errors.push('Point has non-finite coordinates');
    }
    if (Math.abs(lat) > 90) errors.push('Point latitude out of range');
    if (Math.abs(lon) > 180) errors.push('Point longitude out of range');
    return errors;
  }

  const polys =
    geometry.type === 'Polygon' ? [geometry.coordinates] : geometry.coordinates;

  for (const polygon of polys) {
    const exterior = polygon[^0];
    if (!exterior || exterior.length < 4) {
      errors.push('Polygon exterior needs at least 3 vertices + close');
      continue;
    }
    if (!isClosedRing(exterior)) {
      errors.push('Polygon exterior ring must be closed');
    }
  }

  return errors;
}

export function assertValidGeometry(geometry: TerraGeometry): void {
  const errors = validateGeometry(geometry);
  if (errors.length) {
    throw new Error(`Invalid geometry: ${errors.join('; ')}`);
  }
}
```

```ts
import centroid from '@turf/centroid';
import type { TerraGeometry } from '@/entities/types';

/** Returns [lon, lat] suitable for label anchors */
export function geometryCentroid(geometry: TerraGeometry): [number, number] {
  if (geometry.type === 'Point') {
    return [geometry.coordinates[^0], geometry.coordinates[^1]];
  }
  const feature = {
    type: 'Feature' as const,
    properties: {},
    geometry,
  };
  const c = centroid(feature);
  return c.geometry.coordinates as [number, number];
}
```

### `src/state/history/HistoryStack.ts`

```ts
export interface Command {
  name: string;
  execute: () => void;
  undo: () => void;
}

export class HistoryStack {
  private past: Command[] = [];
  private future: Command[] = [];
  private readonly max = 100;

  execute(cmd: Command): void {
    cmd.execute();
    this.past.push(cmd);
    if (this.past.length > this.max) this.past.shift();
    this.future = [];
  }

  undo(): void {
    const cmd = this.past.pop();
    if (!cmd) return;
    cmd.undo();
    this.future.push(cmd);
  }

  redo(): void {
    const cmd = this.future.pop();
    if (!cmd) return;
    cmd.execute();
    this.past.push(cmd);
  }

  clear(): void {
    this.past = [];
    this.future = [];
  }
}

export const history = new HistoryStack();
```

```ts
export function createTerraforgeViewer(container: HTMLElement): Viewer {

    return viewer;
  }

  // Intentionally NO Ion.defaultAccessToken — offline stylized globe
  viewer = new Viewer(container, {

    timeline: false,
    baseLayerPicker: false,
    geocoder: false,
    homeButton: false,
    sceneModePicker: false,
    navigationHelpButton: false,
    fullscreenButton: false,
    infoBox: false,
    selectionIndicator: true,
    creditContainer: document.createElement('div'),
    terrainProvider: new EllipsoidTerrainProvider(),
    sceneMode: SceneMode.SCENE3D,
  });

  // Remove default imagery; solid ocean-like globe
  viewer.imageryLayers.removeAll();
  viewer.scene.globe.baseColor = Color.fromCssColorString('#0c4a6e');
  viewer.scene.backgroundColor = Color.fromCssColorString('#020617');
  viewer.scene.globe.enableLighting = false;
  viewer.scene.fog.enabled = true;
  viewer.scene.skyAtmosphere.show = true;

  viewer.camera.setView({
    destination: Cartesian3.fromDegrees(0, 20, 12_000_000),
    orientation: {
      heading: 0,
      pitch: CesiumMath.toRadians(-90),
      roll: 0,
    },
  });

  return viewer;
}

export function destroyTerraforgeViewer(): void {
  if (viewer && !viewer.isDestroyed()) {
    viewer.destroy();
  }
  viewer = null;
```

### `src/globe/entitySync.ts`

```ts
import {
  Viewer,
  Color,
  Cartesian3,
  HeightReference,
  HorizontalOrigin,
  VerticalOrigin,
  Entity,
} from 'cesium';
import type { TerraEntity } from '@/entities/types';
import { geometryCentroid } from '@/geo/centroid';

const managed = new Map<string, Entity>();

export function syncEntitiesToCesium(
  viewer: Viewer,
  entities: Record<string, TerraEntity>,
  selectedId: string | null,
): void {
  const nextIds = new Set(Object.keys(entities));

  // Remove deleted
  for (const [id, entity] of managed) {
    if (!nextIds.has(id)) {
      viewer.entities.remove(entity);
      managed.delete(id);
    }
  }

  for (const entity of Object.values(entities)) {
    const existing = managed.get(entity.id);
    if (existing) {
      viewer.entities.remove(existing);
      managed.delete(entity.id);
    }
    const cesiumEntity = upsertCesiumEntity(viewer, entity, selectedId === entity.id);
    managed.set(entity.id, cesiumEntity);
  }
}

function upsertCesiumEntity(
  viewer: Viewer,
  entity: TerraEntity,
  selected: boolean,
): Entity {
  const [lon, lat] = geometryCentroid(entity.geometry);
  const fill = Color.fromCssColorString(entity.color).withAlpha(entity.fillOpacity);
  const outline = Color.fromCssColorString(entity.color).withAlpha(selected ? 1 : 0.9);

  if (entity.geometry.type === 'Point') {
    return viewer.entities.add({
      id: entity.id,
      name: entity.name,
      position: Cartesian3.fromDegrees(lon, lat),
      point: {
        pixelSize: selected ? 14 : 10,
        color: fill,
        outlineColor: outline,
        outlineWidth: 2,
        heightReference: HeightReference.CLAMP_TO_GROUND,
      },
      label: {
        text: entity.name,
        font: '14px Inter, sans-serif',
        fillColor: Color.WHITE,
        outlineColor: Color.BLACK,
        outlineWidth: 3,
        style: 2, // FILL_AND_OUTLINE
        verticalOrigin: VerticalOrigin.BOTTOM,
        horizontalOrigin: HorizontalOrigin.CENTER,
        pixelOffset: new Cartesian3(0, -12, 0) as unknown as Cesium.Cartesian2,
        disableDepthTestDistance: Number.POSITIVE_INFINITY,
      },
    });
  }

  // Polygon / MultiPolygon — Phase 2 will refine MultiPolygon
  const hierarchy = polygonHierarchy(entity);
  return viewer.entities.add({
    id: entity.id,
    name: entity.name,
    polygon: {
      hierarchy,
      material: fill,
      outline: true,
      outlineColor: outline,
      outlineWidth: selected ? 3 : 2,
    },
    position: Cartesian3.fromDegrees(lon, lat),
    label: {
      text: entity.name,
      font: '16px Inter, sans-serif',
      fillColor: Color.WHITE,
      outlineColor: Color.BLACK,
      outlineWidth: 3,
      style: 2,
      verticalOrigin: VerticalOrigin.CENTER,
      disableDepthTestDistance: Number.POSITIVE_INFINITY,
    },
  });
}

function polygonHierarchy(entity: TerraEntity): Cesium.PolygonHierarchy {
  const geom = entity.geometry;
  if (geom.type !== 'Polygon' && geom.type !== 'MultiPolygon') {
    return new Cesium.PolygonHierarchy();
  }
  const ring =
    geom.type === 'Polygon'
      ? geom.coordinates[^0]
      : geom.coordinates[^0]?.[^0] ?? [];
  const positions = ring.map(([lng, lt]) => Cartesian3.fromDegrees(lng, lt));
  return new Cesium.PolygonHierarchy(positions);
}

// Re-export namespace bit used above
import * as Cesium from 'cesium';
```

## 6. `docs/DATA_MODEL.md` (short)

```markdown
# Data model

- World.version must be `1`
- Entity ids: ULID or UUID string
- geometry: GeoJSON CRS84 (lon, lat degrees), WGS84
- Polygon rings: exterior closed, CCW preferred
- parentId: null for top-level continents/islands
- colors: #RRGGBB
- images[].blobKey: IndexedDB key (Phase 5)

See `src/entities/types.ts` as source of truth.
```
